cmake_minimum_required(VERSION 3.20)
project(boids-test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(flecs STATIC vendor/flecs.c)
target_include_directories(flecs PUBLIC vendor)
add_library(simulation STATIC src/simulation.cpp)
target_include_directories(simulation PUBLIC src)
target_link_libraries(simulation PUBLIC flecs)
add_executable(simulation_test tests/simulation_test.cpp)
target_link_libraries(simulation_test PRIVATE simulation)
enable_testing()
add_test(NAME simulation COMMAND simulation_test)
